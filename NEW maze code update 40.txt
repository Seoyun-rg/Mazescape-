        function startTimer() {
            const timerElement = document.getElementById('timer-val');
            
            timerInterval = setInterval(() => {
                if (!gameActive) return;

                timeRemaining--;

                let minutes = Math.floor(timeRemaining / 60);
                let seconds = timeRemaining % 60;

                let formattedMinutes = minutes < 10 ? "0" + minutes : minutes;
                let formattedSeconds = seconds < 10 ? "0" + seconds : seconds;

                timerElement.innerText = `${formattedMinutes}:${formattedSeconds}`;

                if (timeRemaining <= 20) {
                    timerElement.classList.add('time-critical');
                }

                if (timeRemaining <= 0) {
                    triggerGameOver("¡TIEMPO AGOTADO! Te has quedado atrapado en el laberinto.");
                }
            }, 1000);
        }

        function shootFireball() {
            if (!cerberusMesh || !fireballModel || isCerberusDead || !isCerberusAwake) return;

            const fireball = fireballModel.clone();
            
            const startPos = cerberusMesh.position.clone();
            const forward = new THREE.Vector3().subVectors(cameraHolder.position, cerberusMesh.position);
            forward.y = 0;
            forward.normalize();

            startPos.addScaledVector(forward, 2.8); 
            startPos.y += 20.0; 

            fireball.position.copy(startPos);

            const targetPos = cameraHolder.position.clone();
            const direction = new THREE.Vector3().subVectors(targetPos, startPos).normalize();

            fireball.lookAt(targetPos);

            fireball.rotateX(FIREBALL_ROTATION.x);
            fireball.rotateY(FIREBALL_ROTATION.y);
            fireball.rotateZ(FIREBALL_ROTATION.z);

            scene.add(fireball);
            activeFireballs.push({
                mesh: fireball,
                direction: direction,
                lifeTime: 0
            });

            playSFX(300, 0.3);
        }

        function checkCollision(moveVector) {
            if (collidableObjects.length === 0) return false;
            const rayDir = moveVector.clone().normalize();
            raycaster.set(cameraHolder.position, rayDir);

            const intersects = raycaster.intersectObjects(collidableObjects, true);
            if (intersects.length > 0 && intersects[0].distance < 0.8) {
                return true;
            }
            return false;
        }

        function checkEnemyCollision(position, moveVector, minDistance = 3.8) {
            if (collidableObjects.length === 0) return false;
            const rayDir = moveVector.clone().normalize();
            const yOffsets = [0.8, 1.8, 2.6];
            const perp = new THREE.Vector3(-rayDir.z, 0, rayDir.x).normalize().multiplyScalar(1.5);
            const xzOffsets = [ new THREE.Vector3(0, 0, 0), perp.clone(), perp.clone().negate() ];

            for (let y of yOffsets) {
                for (let xz of xzOffsets) {
                    const origin = position.clone().add(xz);
                    origin.y += y;
                    enemyRaycaster.set(origin, rayDir);
                    const intersects = enemyRaycaster.intersectObjects(collidableObjects, true);
                    if (intersects.length > 0 && intersects[0].distance < minDistance) {
                        return true;
                    }
                }
            }
            return false;
        }

        function getValidGroundData(position) {
            if (collidableObjects.length === 0) return { y: groundY, isWater: false };
            const rayOrigin = position.clone();
            rayOrigin.y += 20.0;
            downRaycaster.set(rayOrigin, new THREE.Vector3(0, -1, 0));
            
            const intersects = downRaycaster.intersectObjects(collidableObjects, true);
            if (intersects.length > 0) {
                const hit = intersects[0];
                const hitObj = hit.object;
                const name = hitObj.name.toLowerCase();
                
                let isWater = name.includes("water") || name.includes("agua") || name.includes("azul") || name.includes("river");
                
                if (hitObj.material && hitObj.material.color) {
                    const col = hitObj.material.color;
                    if (col.b > col.r + 0.2 && col.b > col.g + 0.2) {
                        isWater = true;
                    }
                }

                return { y: hit.point.y, isWater: isWater };
            }
            return { y: groundY, isWater: false };
        }

        function attackWithWeapon() {
            if (isAttacking || attackDamage === 0) return;
            isAttacking = true;

            playSFX(attackDamage > 30 ? 600 : 400, 0.12);

            if (currentWeaponMesh) {
                anime({
                    targets: currentWeaponMesh.rotation,
                    x: [0.2, 1.3, 0.2],

                    z: [0.2, -0.4, 0.2],
                    duration: 220,
                    easing: 'easeOutQuad',
                    complete: () => { isAttacking = false; }
                });
            } else {
                setTimeout(() => { isAttacking = false; }, 220);
            }

            if (minotaurMesh && !isMinotaurDead) {
                const minotaurCenterPos = minotaurMesh.position.clone();
                minotaurCenterPos.y += 2.0; 

                const dist = cameraHolder.position.distanceTo(minotaurCenterPos);
                if (dist < 8.5) {
                    playSFX(120, 0.25);
                    const pushDir = new THREE.Vector3().subVectors(minotaurMesh.position, cameraHolder.position).normalize();
                    
                    if (!checkEnemyCollision(minotaurMesh.position, pushDir, 3.8)) {
                        minotaurMesh.position.addScaledVector(pushDir, 2.2);
                        minotaurMesh.position.y = getValidGroundData(minotaurMesh.position).y;
                    }

                    minotaurHP -= attackDamage;
                    updateHealthBar();

                    minotaurMesh.traverse((child) => {
                        if (child.isMesh && child.material) {
                            const origColor = child.material.color.getHex();
                            child.material.color.setHex(0xFF0000);
                            setTimeout(() => { if (child.material) child.material.color.setHex(origColor); }, 150);
                        }
                    });

                    if (minotaurHP <= 0) killMinotaur();
                }
            }

            if (cerberusMesh && isCerberusAwake && !isCerberusDead) {
                const distCerberus = cameraHolder.position.distanceTo(cerberusMesh.position);
                if (distCerberus < 8.5) {
                    playSFX(100, 0.25);
                    const pushDir = new THREE.Vector3().subVectors(cerberusMesh.position, cameraHolder.position).normalize();
                    
                    if (!checkEnemyCollision(cerberusMesh.position, pushDir, 3.8)) {
                        cerberusMesh.position.addScaledVector(pushDir, 0.8);
                        cerberusMesh.position.y = getValidGroundData(cerberusMesh.position).y - 2.8;
                    }

                    cerberusHP -= attackDamage;
                    updateCerberusHealthBar();

                    cerberusMesh.traverse((child) => {
                        if (child.isMesh && child.material) {
                            const origColor = child.material.color.getHex();
                            child.material.color.setHex(0xFF0000);
                            setTimeout(() => { if (child.material) child.material.color.setHex(origColor); }, 150);
                        }
                    });

                    if (cerberusHP <= 0) killCerberus();
                }
            }
        }

        function killMinotaur() {
            isMinotaurDead = true;
            if (minotaurHealthBarMesh) minotaurHealthBarMesh.visible = false;
            playSFX(80, 0.6);

            scene.remove(minotaurMesh);

            hasKey = true;
            playSFX(1000, 0.3);

            const keyHud = document.getElementById('key-status');
            keyHud.innerText = "LLAVE: POSEÍDA 🗝️";
            keyHud.style.color = "#00FFFF";

            agregarAlInventario('key', ICONS.key);
        }

        function killCerberus() {
            isCerberusDead = true;
            if (cerberusHealthBarMesh) cerberusHealthBarMesh.visible = false;
            playSFX(60, 0.8);

            anime({
                targets: cerberusMesh.rotation,
                z: Math.PI / 2,
                duration: 900,
                easing: 'easeOutBounce',
                complete: () => {
                    anime({
                        targets: cerberusMesh.position,
                        y: cerberusMesh.position.y - 1.5,
                        duration: 1500,
                        easing: 'easeInQuad',
                        complete: () => {
                            scene.remove(cerberusMesh);
                        }
                    });
                }
            });
        }

        function playSFX(freq, duration) {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + duration);
            } catch(e){}
        }

        function animate() {
            requestAnimationFrame(animate);
            const delta = clock.getDelta();

            if (mixer && !isMinotaurDead) mixer.update(delta);
            if (cerberusMixer && isCerberusAwake && !isCerberusDead) cerberusMixer.update(delta);

            if (gameActive) {
                const speed = 40.0;
                const moveVector = new THREE.Vector3();

                if (moveForward) moveVector.z -= 1;
                if (moveBackward) moveVector.z += 1;
                if (moveLeft) moveVector.x -= 1;
                if (moveRight) moveVector.x += 1;

                if (moveVector.lengthSq() > 0) {
                    moveVector.normalize();
                    moveVector.applyAxisAngle(new THREE.Vector3(0, 1, 0), yaw);

                    if (!checkCollision(moveVector)) {
                        cameraHolder.position.addScaledVector(moveVector, speed * delta);
                    }
                }

                cameraHolder.position.y = groundY + 1.8;

                const distFromCenter = Math.hypot(cameraHolder.position.x, cameraHolder.position.z);
                const victoryRadius = mazeRadius * victoryRadiusMultiplier;
                
                if (distFromCenter > victoryRadius) {
                    triggerVictory();
                }

                if (hasKey && !isDoorUnlocked && doorMeshes.length > 0) {
                    doorMeshes.forEach((doorMesh) => {
                        const doorWorldPos = new THREE.Vector3();
                        doorMesh.getWorldPosition(doorWorldPos);

                        if (cameraHolder.position.distanceTo(doorWorldPos) < 4.0) {
                            unlockAndOpenDoor();
                        }
                    });
                }

                if (cerberusMesh && !isCerberusAwake && !isCerberusDead) {
                    if (cameraHolder.position.distanceTo(cerberusMesh.position) < 8.0) {
                        wakeUpCerberus();
                    }
                }

                if (swordPickupMesh && !hasSword) {
                    swordPickupMesh.rotation.y += 0.01;

                    if (cameraHolder.position.distanceTo(swordPickupMesh.position) < 2.0) {
                        hasSword = true;
                        scene.remove(swordPickupMesh);
                        playSFX(900, 0.3);

                        agregarAlInventario('sword', ICONS.sword);

                        const swordSlotIndex = inventarioSlots.indexOf('sword');
                        if (swordSlotIndex !== -1) seleccionarSlot(swordSlotIndex);
                    }
                }

                if (hammerPickupMesh && !hasHammer) {
                    hammerPickupMesh.rotation.y += 0.01;

                    if (cameraHolder.position.distanceTo(hammerPickupMesh.position) < 2.0) {
                        hasHammer = true;
                        scene.remove(hammerPickupMesh);
                        playSFX(950, 0.3);

                        agregarAlInventario('hammer', ICONS.hammer);

                        const hammerSlotIndex = inventarioSlots.indexOf('hammer');
                        if (hammerSlotIndex !== -1) seleccionarSlot(hammerSlotIndex);
                    }
                }

                if (minotaurHealthBarMesh && !isMinotaurDead) minotaurHealthBarMesh.lookAt(cameraHolder.position);
                if (cerberusHealthBarMesh && !isCerberusDead) cerberusHealthBarMesh.lookAt(cameraHolder.position);

                if (minotaurMesh && !isMinotaurDead) {
                    const minoSpeed = 30.0;
                    const dirToPlayer = new THREE.Vector3().subVectors(cameraHolder.position, minotaurMesh.position);
                    dirToPlayer.y = 0;
                    
                    const distance = dirToPlayer.length();
                    dirToPlayer.normalize();

                    if (distance > 2.0) {
                        let nextPos = minotaurMesh.position.clone().addScaledVector(dirToPlayer, minoSpeed * delta);
                        let groundData = getValidGroundData(nextPos);

                        if (groundData.isWater || checkEnemyCollision(minotaurMesh.position, dirToPlayer, 3.8)) {
                            const angleLeft = dirToPlayer.clone().applyAxisAngle(new THREE.Vector3(0, 1, 0), Math.PI / 3);
                            const angleRight = dirToPlayer.clone().applyAxisAngle(new THREE.Vector3(0, 1, 0), -Math.PI / 3);
                            
                            let posLeft = minotaurMesh.position.clone().addScaledVector(angleLeft, minoSpeed * delta);
                            let posRight = minotaurMesh.position.clone().addScaledVector(angleRight, minoSpeed * delta);

                            let dataLeft = getValidGroundData(posLeft);
                            let dataRight = getValidGroundData(posRight);

                            if (!dataLeft.isWater && !checkEnemyCollision(minotaurMesh.position, angleLeft, 3.2)) {
                                minotaurMesh.position.copy(posLeft);
                            } else if (!dataRight.isWater && !checkEnemyCollision(minotaurMesh.position, angleRight, 3.2)) {
                                minotaurMesh.position.copy(posRight);
                            } else {
                                minotaurMesh.position.copy(nextPos);
                            }
                        } else {
                            minotaurMesh.position.copy(nextPos);
                        }

                        minotaurMesh.position.y = getValidGroundData(minotaurMesh.position).y;
                        minotaurMesh.lookAt(cameraHolder.position.x, minotaurMesh.position.y, cameraHolder.position.z);
                    } else {
                        playerHP -= 15 * delta;
                        updatePlayerHearts();
                        triggerDamageFlash();
                        if (playerHP <= 0) {
                            triggerGameOver("¡HAS SIDO ATRAPADO Y ELIMINADO POR EL MINOTAURO!");
                        }
                    }
                }

                if (cerberusMesh && isCerberusAwake && !isCerberusDead) {
                    const cerberusSpeed = 10.0;
                    const dirToPlayer = new THREE.Vector3().subVectors(cameraHolder.position, cerberusMesh.position);
                    dirToPlayer.y = 0;
                    
                    const distance = dirToPlayer.length();
                    dirToPlayer.normalize();

                    cerberusAttackTimer += delta;
                    if (cerberusAttackTimer > 2.0) {
                        shootFireball();
                        cerberusAttackTimer = 0;
                    }

                    if (distance > 2.0) {
                        let nextPos = cerberusMesh.position.clone().addScaledVector(dirToPlayer, cerberusSpeed * delta);
                        let groundData = getValidGroundData(nextPos);

                        if (groundData.isWater || checkEnemyCollision(cerberusMesh.position, dirToPlayer, 3.8)) {
                            const angleLeft = dirToPlayer.clone().applyAxisAngle(new THREE.Vector3(0, 1, 0), Math.PI / 3);
                            const angleRight = dirToPlayer.clone().applyAxisAngle(new THREE.Vector3(0, 1, 0), -Math.PI / 3);
                            
                            let posLeft = cerberusMesh.position.clone().addScaledVector(angleLeft, cerberusSpeed * delta);
                            let posRight = cerberusMesh.position.clone().addScaledVector(angleRight, cerberusSpeed * delta);

                            let dataLeft = getValidGroundData(posLeft);
                            let dataRight = getValidGroundData(posRight);

                            if (!dataLeft.isWater && !checkEnemyCollision(cerberusMesh.position, angleLeft, 3.2)) {
                                cerberusMesh.position.copy(posLeft);
                            } else if (!dataRight.isWater && !checkEnemyCollision(cerberusMesh.position, angleRight, 3.2)) {
                                cerberusMesh.position.copy(posRight);
                            }
                        } else {
                            cerberusMesh.position.copy(nextPos);
                        }

                        cerberusMesh.position.y = getValidGroundData(cerberusMesh.position).y;
                        cerberusMesh.lookAt(cameraHolder.position.x, cerberusMesh.position.y, cameraHolder.position.z);
                    } else {
                        playerHP -= 20 * delta;
                        updatePlayerHearts();
                        triggerDamageFlash();
                        if (playerHP <= 0) {
                            triggerGameOver("¡CERBERO TE HA DEVORADO EN EL CENTRO DEL LABERINTO!");
                        }
                    }
                }

                // ACTUALIZACIÓN Y DETECCIÓN DE COLISIONES DE FIREBALLS
                for (let i = activeFireballs.length - 1; i >= 0; i--) {
                    const fb = activeFireballs[i];
                    fb.lifeTime += delta;
                    const fbSpeed = 18.0;

                    // Mover bola de fuego
                    fb.mesh.position.addScaledVector(fb.direction, fbSpeed * delta);

                    // 1. Detección de colisión con muros/mapa
                    if (collidableObjects.length > 0) {
                        fireballRaycaster.set(fb.mesh.position, fb.direction);
                        const wallHits = fireballRaycaster.intersectObjects(collidableObjects, true);

                        if (wallHits.length > 0 && wallHits[0].distance < 1.2) {
                            // Impacto con pared/laberinto
                            playSFX(180, 0.2); // Sonido de impacto
                            scene.remove(fb.mesh);
                            activeFireballs.splice(i, 1);
                            continue;
                        }
                    }

                    // 2. Detección de colisión con el Jugador
                    const distToPlayer = fb.mesh.position.distanceTo(cameraHolder.position);
                    if (distToPlayer < 1.5) {
                        playerHP -= 20;
                        updatePlayerHearts();
                        triggerDamageFlash();
                        playSFX(150, 0.4);

                        scene.remove(fb.mesh);
                        activeFireballs.splice(i, 1);

                        if (playerHP <= 0) {
                            triggerGameOver("¡HAS SIDO IMPACTADO Y INCINERADO POR EL CERBERO!");
                        }
                        continue;
                    }

                    // 3. Tiempo de vida máximo de la bola de fuego
                    if (fb.lifeTime > 6.0) {
                        scene.remove(fb.mesh);
                        activeFireballs.splice(i, 1);
                    }
                }

                gemsArray.forEach((gem, idx) => {
                    gem.rotation.y += 0.02;
                    if (cameraHolder.position.distanceTo(gem.position) < 1.4) {
                        scene.remove(gem);
                        gemsArray.splice(idx, 1);
                        gemsCollected++;
                        playSFX(800, 0.15);
                        document.getElementById('gems-val').innerText = gemsCollected;
                    }
                });
            }

            renderer.render(scene, camera);
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }
